Буду разворачивать кластер из 3 узлов в `Yandex Cloud` на базе `Pacemaker/Corosync/DRBD`.

# Terraform

Создаю файл конфигурации Terraform CLI:
```bash
mkdir -p ~/.terraform.d && nano ~/.terraformrc
```
```yaml
provider_installation {
    network_mirror {
        url = "https://terraform-mirror.yandexcloud.net/"
        include = ["registry.terraform.io/*/*"]
}
    direct {
        exclude = ["registry.terraform.io/*/*"]
    }
}
```

Создаю файл настроек провайдера:
```bash
nano providers.tf
```
```yaml
terraform {
  required_providers {
    yandex = {
      source = "yandex-cloud/yandex"
    }
  }
  required_version = ">= 0.13"
}

provider "yandex" {
  service_account_key_file = "key.json"
  cloud_id  = "b1gmqqn8huu6s39112te"
  folder_id = "b1gk5pe5h3541d01896e"
  zone      = "ru-central1-a"
}
```

Создаю файл переменных:
```bash
nano variables.tf
```
```yaml
variable "cloud_id" {
  description = "Yandex Cloud ID"
  type        = string
}

variable "folder_id" {
  description = "Yandex Cloud Folder ID"
  type        = string
}

variable "zone" {
  description = "Availability zone"
  type        = string
  default     = "ru-central1-a"
}

variable "public_key_path" {
  description = "Path to public SSH key"
  type        = string
  default     = "~/.ssh/id_rsa.pub"
}
```

Создаю файл значений переменных:
```bash
nano terraform.tfvars
```
```yaml
cloud_id = "b1gmqqn8huu6s39112te"
folder_id = "b1gk5pe5h3541d01896e"
zone = "ru-central1-a"
public_key_path = "/home/zotac/.ssh/id_rsa.pub"
```

Создаю файл конфигурации ВМ:
```bash
nano main.tf
```
```yaml
resource "yandex_vpc_network" "default" {
  name = "my-network"
}

resource "yandex_vpc_subnet" "default" {
  name           = "my-subnet"
  zone           = var.zone
  network_id     = yandex_vpc_network.default.id
  v4_cidr_blocks = ["192.168.10.0/24"]
}

data "local_file" "public_key" {
  filename = var.public_key_path
}

data "yandex_compute_image" "ubuntu" {
  family    = "ubuntu-24-04-lts"
  folder_id = "standard-images"
}

resource "yandex_compute_disk" "additional" {
  count = 3

  name     = "additional-ssd-${count.index + 1}"
  type     = "network-ssd"
  zone     = var.zone
  size     = 30
}

resource "yandex_compute_instance" "vm" {
  count = 3

  name        = "vm-${count.index + 1}"
  platform_id = "standard-v3"
  zone        = var.zone

  resources {
    cores  = 4
    memory = 8
  }

  boot_disk {
    initialize_params {
      image_id = data.yandex_compute_image.ubuntu.id
      size     = 20
      type     = "network-ssd"
    }
  }

  secondary_disk {
    disk_id = yandex_compute_disk.additional[count.index].id
  }

  network_interface {
    subnet_id = yandex_vpc_subnet.default.id
    nat       = true
  }

  metadata = {
    ssh-keys = "ubuntu:${data.local_file.public_key.content}"
  }
}

output "vm1_ip" {
  value = yandex_compute_instance.vm[0].network_interface.0.nat_ip_address
}

output "vm2_ip" {
  value = yandex_compute_instance.vm[1].network_interface.0.nat_ip_address
}

output "vm3_ip" {
  value = yandex_compute_instance.vm[2].network_interface.0.nat_ip_address
}
```

# Ansible

Создаю файл конфига:
```
nano ansible.cfg
```
```
[defaults]
inventory = inventory.ini
host_key_checking = False
timeout = 30
remote_tmp = /tmp/ansible-${USER}

[ssh_connection]
pipelining = True
```

Создание файла инвентаря:
```bash
nano inventory.ini
```
```
[postgres_cluster]
node1 ansible_host=93.77.187.217 innner_ip=192.168.10.9
node2 ansible_host=89.169.146.111 innner_ip=192.168.10.22
node3 ansible_host=89.169.137.10 innner_ip=192.168.10.27

[primary]
node1

[postgres_cluster:vars]
ansible_user=ubuntu
```

Плейбук установки PostgreSQL:
```
nano install_postgresql18.yml
```
```yaml
---
- name: Install PostgreSQL 18
  hosts: postgres_cluster
  become: yes
  gather_facts: yes

  tasks:
    - name: Add PostgreSQL GPG key
      apt_key:
        url: https://www.postgresql.org/media/keys/ACCC4CF8.asc
        state: present

    - name: Add PostgreSQL repository
      apt_repository:
        repo: "deb https://apt.postgresql.org/pub/repos/apt {{ ansible_distribution_release }}-pgdg main"
        state: present

    - name: Install PostgreSQL 18
      apt:
        name: postgresql-18
        state: present
```
