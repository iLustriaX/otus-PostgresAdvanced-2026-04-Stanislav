  name     = "additional-ssd-${count.index + 1}"
  type     = "network-ssd"
  zone     = var.zone

  count = 3

  name        = "vm-${count.index + 1}"
  platform_id = "standard-v3"

    cores  = 4
    memory = 8
  }

  boot_disk {
    initialize_params {
      image_id = data.yandex_compute_image.ubuntu.id
      size     = 20
      type     = "network-ssd"
    }
  }

  secondary_disk {
    disk_id = yandex_compute_disk.additional[count.index].id
  }

  network_interface {
    subnet_id = yandex_vpc_subnet.default.id
    nat       = true
  }

  metadata = {
    ssh-keys = "ubuntu:${data.local_file.public_key.content}"
  }

output "vm1_ip" {
  value = yandex_compute_instance.vm[0].network_interface.0.nat_ip_address

Плейбук установки необходимых пакетов:
```
nano install_common.yml
```
```yaml
---
- name: Prepare all nodes for HA cluster
  hosts: postgres_cluster
  become: yes
  gather_facts: yes

  tasks:
    - name: Update apt cache
      apt:
        update_cache: yes
        cache_valid_time: 3600

    - name: Install base packages
      apt:
        name:
          - pacemaker
          - corosync
          - pcs
          - drbd-utils
          - chrony
        state: present

    - name: Enable and start chrony
      service:
        name: chrony
        state: started
        enabled: yes

    - name: Set hostname
      hostname:
        name: "{{ inventory_hostname }}"

    - name: Remove old host entries for nodes
      lineinfile:
        path: /etc/hosts
        regexp: '^{{ item }}\s'
        state: absent
      loop: "{{ groups['postgres_cluster'] }}"

    - name: Add correct host entries for nodes (internal IPs)
      lineinfile:
        path: /etc/hosts
        line: "{{ hostvars[item].drbd_ip }} {{ item }}"
        state: present
      loop: "{{ groups['postgres_cluster'] }}"

    - name: Stop and disable PostgreSQL (if installed)
      service:
        name: postgresql
        state: stopped
        enabled: no
      ignore_errors: yes
```

Плейбук настройки drbd:
```
nano configure_drbd.yml
```
```yaml
---
- name: Configure DRBD
  hosts: postgres_cluster
  become: yes
  gather_facts: yes

  tasks:
    - name: Remove old DRBD config (if any)
      file:
        path: /etc/drbd.d/postgres.res
        state: absent

    - name: Stop DRBD (if running)
      command: drbdadm down postgres
      ignore_errors: yes

    - name: Create correct DRBD configuration (hostnames, not IPs)
      copy:
        dest: /etc/drbd.d/postgres.res
        content: |
          resource postgres {
            protocol C;
            device /dev/drbd0;
            disk /dev/vdb;
            meta-disk internal;

          {% for host in groups['postgres_cluster'] %}
            on {{ host }} {
              address {{ hostvars[host].drbd_ip }}:7788;
            }
          {% endfor %}
          }
      # Ansible выполнит шаблонизацию содержимого, так как использует Jinja2

    - name: Initialize DRBD metadata on all nodes (force overwrite)
      command: drbdadm create-md --force postgres

    - name: Bring up DRBD on all nodes
      command: drbdadm up postgres

    - name: Wait for DRBD to become connected (optional)
      pause:
        seconds: 5

    - name: Make primary node primary (only on node1)
      command: drbdadm primary --force postgres
      when: inventory_hostname == groups['primary'][0]

    - name: Create filesystem on /dev/drbd0 (only on primary, if not exists)
      filesystem:
        dev: /dev/drbd0
        fstype: ext4
        force: no
      when: inventory_hostname == groups['primary'][0]

    - name: Create mount point for PostgreSQL data on all nodes
      file:
        path: /var/lib/postgresql/18/main
        state: directory
        owner: postgres
        group: postgres
        mode: 0755
```
