# Terraform

Создаю файл конфигурации Terraform CLI:
```bash
mkdir -p ~/.terraform.d && nano ~/.terraformrc
```
```yaml
provider_installation {
    network_mirror {
        url = "https://terraform-mirror.yandexcloud.net/"
        include = ["registry.terraform.io/*/*"]
}
    direct {
        exclude = ["registry.terraform.io/*/*"]
    }
}
```

Создаю файл настроек провайдера:
```bash
nano providers.tf
```
```yaml
terraform {
  required_providers {
    yandex = {
      source = "yandex-cloud/yandex"
    }
  }
  required_version = ">= 0.13"
}

provider "yandex" {
  service_account_key_file = "key.json"
  cloud_id  = "b1gmqqn8huu6s39112te"
  folder_id = "b1gk5pe5h3541d01896e"
  zone      = "ru-central1-a"
}
```

Создаю файл переменных:
```bash
nano variables.tf
```
```yaml
variable "cloud_id" {
  description = "Yandex Cloud ID"
  type        = string
}

variable "folder_id" {
  description = "Yandex Cloud Folder ID"
  type        = string
}

variable "zone" {
  description = "Availability zone"
  type        = string
  default     = "ru-central1-a"
}

  description = "Path to public SSH key"
  type        = string

}
```

## Плейбуки
```
etcd.yml[]()
```
```
- name: Install and configure ETCD cluster
  hosts: postgresql
  become: yes
  vars:
    etcd_cluster_name: "my-etcd"
    etcd_initial_cluster: |
      {% for host in groups['postgresql'] %}
      {{ hostvars[host].inventory_hostname }}=http://{{ hostvars[host].private_ip }}:2380{% if not loop.last %},{% endif %}
      {% endfor %}
  tasks:
    - name: Update apt cache
      apt:
        update_cache: yes
        cache_valid_time: 3600

    - name: Install etcd-server
      apt:
        name: etcd-server
        state: present

    - name: Create etcd config
      template:
        src: ../templates/etcd.conf.j2
        dest: /etc/default/etcd
      notify: restart etcd

    - name: Start and enable etcd
      systemd:
        name: etcd
        state: started
        enabled: yes

  handlers:
    - name: restart etcd
      systemd:
        name: etcd
        state: restarted
```

```
patroni_postgresql.yaml
```
```
- name: Install Patroni and PostgreSQL 18
  hosts: postgresql
  become: yes
  vars:
    patroni_cluster_name: "my-cluster"
    patroni_etcd_hosts: "{{ groups['postgresql'] | map('extract', hostvars, 'private_ip') | map('regex_replace', '^(.*)$', '\\1:2379') | join(',') }}"
    patroni_data_dir: /var/lib/postgresql/18/main
    patroni_bin_dir: /usr/lib/postgresql/18/bin
  tasks:
    - name: Update apt cache
      apt:
        update_cache: yes
        cache_valid_time: 3600

    - name: Add PostgreSQL GPG key
      apt_key:
        url: https://www.postgresql.org/media/keys/ACCC4CF8.asc
        state: present

    - name: Add PostgreSQL repository
      apt_repository:
        repo: "deb https://apt.postgresql.org/pub/repos/apt {{ ansible_distribution_release }}-pgdg main"
        state: present

    - name: Install PostgreSQL 18 and Patroni dependencies
      apt:
        name:
          - postgresql-18
          - patroni
          - python3-psycopg2
        state: present

    - name: Stop PostgreSQL service (if running)
      systemd:
        name: postgresql
        state: stopped
      ignore_errors: yes

    - name: Disable PostgreSQL service from starting automatically
      systemd:
        name: postgresql
        enabled: no
      ignore_errors: yes

    - name: Remove default PostgreSQL cluster (if exists)
      command: pg_dropcluster --stop 18 main
      register: drop_cluster
      failed_when: false
      changed_when: drop_cluster.rc == 0
      ignore_errors: yes

    - name: Ensure Patroni data directory is clean
      file:
        path: "{{ patroni_data_dir }}"
        state: absent

    - name: Create Patroni config directory
      file:
        path: /etc/patroni
        state: directory
        mode: 0755

    - name: Deploy Patroni config
      template:
        src: ../templates/patroni_postgresql.yml.j2
        dest: /etc/patroni/config.yml
      notify: restart patroni

    - name: Start and enable Patroni
      systemd:
        name: patroni
        state: started
        enabled: yes

  handlers:
    - name: restart patroni
      systemd:
        name: patroni
        state: restarted
```

```
haproxy.cfg.j2
```
```
global
    log /dev/log local0
    maxconn 4096
    user haproxy
    group haproxy
    daemon

defaults
    log global
    mode tcp
    option tcplog
    option dontlognull
    timeout connect 5s
    timeout client 50s
    timeout server 50s

frontend postgres-front
    bind *:5432
    default_backend postgres-back

backend postgres-back
    option httpchk GET /primary
    http-check expect status 200
    default-server inter 3s fall 3 rise 2 on-marked-down shutdown-sessions
    {% for ip in patroni_nodes_ips %}
    server postgres-{{ loop.index }} {{ ip }}:5432 check port 8008
    {% endfor %}

listen stats
    bind *:7000
    stats enable
    stats uri /stats
    stats realm HAProxy\ Statistics
    stats auth {{ haproxy_stats_user }}:{{ haproxy_stats_pass }}
```
